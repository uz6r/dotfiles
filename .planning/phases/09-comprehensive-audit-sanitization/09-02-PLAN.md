```bash
#!/usr/bin/env bash
# bin/audit-secrets — Scan for API keys, tokens, credentials
# Usage:
#   ./audit-secrets           # REPORT mode: print findings, exit 0
#   ./audit-secrets --strict  # STRICT mode: print findings, exit 1 if any found
#   STRICT=1 ./audit-secrets  # Alternative: env var
set -euo pipefail
STRICT="${STRICT:-0}"
[[ "${1:-}" == "--strict" ]] && STRICT=1

total_findings=0
print_finding() {
  local pattern="$1" count="$2"
  if [ "$count" -gt 0 ]; then
    echo "  ⚠️  $pattern → ${count} match(es)"
    total_findings=$((total_findings + 1))
  fi
}

scan() {
  local pattern="$1" label="$2" exclude="$3"
  local count
  count=$(rg --no-heading -c "$pattern" --hidden -g '!.git' -g '!.planning' -g '!private/' -g '!.gitleaks.toml' . 2>/dev/null | wc -l || echo 0)
  if [ "$count" -gt 0 ]; then
    echo "⚠️  ${label}:"
    rg --no-heading -n "$pattern" --hidden -g '!.git' -g '!.planning' -g '!private/' -g '!.gitleaks.toml' . 2>/dev/null | head -20
    echo ""
    print_finding "$label" "$count"
  fi
}

echo "=== Secret Scan ==="
SCAN_PATTERNS=(
  '-----BEGIN (RSA|OPENSSH|DSA|EC|PGP) PRIVATE KEY-----:SSH private key'
  'psk\s*=\s*\S+:WiFi Pre-Shared Key'
  '(api[_-]?key|api[_-]?secret|access[_-]?key|secret[_-]?key)\s*[:=]\s*\S+:API key/secret'
  '(ghp_|gho_|ghu_|ghs_|ghr_)[A-Za-z0-9_]{36,}:GitHub token'
  'sk-[a-zA-Z0-9]{20,}:Stripe/OpenAI key pattern'
  'AKIA[0-9A-Z]{16}:AWS access key'
)
for entry in "${SCAN_PATTERNS[@]}"; do
  pattern="${entry%%:*}"
  label="${entry##*:}"
  scan "$pattern" "$label"
done

if [ "$total_findings" -eq 0 ]; then
  echo "✅ No potential secrets detected"
  exit 0
fi
echo ""
echo "Total: $total_findings pattern(s) with matches"
if [ "$STRICT" -eq 1 ]; then
  echo "❌ Secret scan FAILED (strict mode)"
  exit 1
fi
echo "⚠️  Audit found potential issues (non-blocking in REPORT mode)"
exit 0
```

2. **Create `bin/audit-binaries`** — Dual-mode binary detector:

```bash
#!/usr/bin/env bash
# bin/audit-binaries — Find binary/large files committed to repo
set -euo pipefail
STRICT="${STRICT:-0}"
[[ "${1:-}" == "--strict" ]] && STRICT=1

echo "=== Binary Scan ==="
found=0

# Check for ELF/Mach-O binaries in tracked or untracked files
while IFS= read -r -d '' f; do
  filetype=$(file -b "$f" 2>/dev/null || echo "unknown")
  if echo "$filetype" | grep -qiE 'ELF|Mach-O|executable|binary'; then
    size=$(stat -c%s "$f" 2>/dev/null || stat -f%z "$f" 2>/dev/null || echo "?")
    echo "⚠️  Binary detected: $f ($filetype, ${size} bytes)"
    found=1
  fi
done < <(find . -path ./.git -prune -o -type f -print0 2>/dev/null)

# Check for files > 1MB that might be unintended binaries
while IFS= read -r -d '' f; do
  size=$(stat -c%s "$f" 2>/dev/null || stat -f%z "$f" 2>/dev/null || echo "0")
  if [ "$size" -gt 1048576 ] 2>/dev/null; then
    # Check if it's not in file-type exclusion list
    if file -b "$f" 2>/dev/null | grep -qiE 'archive|compressed|image|audio|video'; then
      :  # skip known large types
    else
      echo "⚠️  Large file: $f (${size} bytes)"
      found=1
    fi
  fi
done < <(find . -path ./.git -prune -o -type f -print0 2>/dev/null)

if [ "$found" -eq 0 ]; then
  echo "✅ No binaries detected"
  exit 0
fi
if [ "$STRICT" -eq 1 ]; then
  echo "❌ Binary scan FAILED (strict mode)"
  exit 1
fi
exit 0
```

#!/usr/bin/env bash
# bin/audit-stale — Find stale configs, empty dirs, dead references

[[ "${1:-}" == "--strict" ]] && STRICT=1

echo "=== Stale Config Scan ==="

stow_dirs=$(ls -d */ 2>/dev/null | tr -d '/')
target_dirs="zsh git nvim bin tmux"

    .git|.github|.githooks|.planning|private|scripts|node_modules) continue ;;
  esac
  if ! echo "$target_dirs" | grep -q "$dir"; then
    echo "⚠️  Directory '$dir' exists but is not a stow target"
    found=1
  fi
done

if [ "$found" -eq 0 ]; then
  echo "✅ No stale config detected"
  exit 0

if [ "$STRICT" -eq 1 ]; then
  echo "❌ Stale config scan FAILED (strict mode)"
